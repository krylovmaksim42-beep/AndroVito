Готовый код бота (Python + aiogram 3)

Сохраните этот HTML-файл на хостинг с HTTPS (например, GitHub Pages или Vercel), вставьте ссылку в WEBAPP_URL ниже и запустите бота. 

import asyncio
from aiogram import Bot, Dispatcher, types
from aiogram.filters import CommandStart
from aiogram.types import InlineKeyboardMarkup, InlineKeyboardButton, WebAppInfo

BOT_TOKEN = "ВАШ_ТОКЕН_ОТ_BOTFATHER"
WEBAPP_URL = "https://ваш-домен.ru/index.html"

bot = Bot(token=BOT_TOKEN)
dp = Dispatcher()

@dp.message(CommandStart())
async def start_cmd(message: types.Message):
    kb = InlineKeyboardMarkup(inline_keyboard=[
        [
            InlineKeyboardButton(text="🚀 Разместить", web_app=WebAppInfo(url=f"{WEBAPP_URL}?action=create")),
            InlineKeyboardButton(text="📱 Приложение", web_app=WebAppInfo(url=WEBAPP_URL))
        ],
        [
            InlineKeyboardButton(text="📋 Мои лоты", web_app=WebAppInfo(url=f"{WEBAPP_URL}?tab=profile")),
            InlineKeyboardButton(text="👤 Профиль", web_app=WebAppInfo(url=f"{WEBAPP_URL}?tab=profile"))
        ],
        [
            InlineKeyboardButton(text="❓ Помощь", callback_data="help"),
            InlineKeyboardButton(text="📢 Канал", url="https://t.me/telegram")
        ]
    ])
    
    text = (
        f"👋 Привет, {message.from_user.first_name}!\n\n"
        "CyberTrade — купи-продай технику без посредников.\n\n"
        "📦 Выложи первый лот — это займёт минуту 👇"
    )
    await message.answer(text, parse_mode="HTML", reply_markup=kb)

async def main():
    await dp.start_polling(bot)

if __name__ == "__main__":
    asyncio.run(main())
